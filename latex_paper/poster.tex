\begin{column}{\colwidth}

  \begin{block}{Background \& Motivation}
    \vspace{0.5ex}
    Astronauts face a \textbf{$\sim$5$\times$ elevated risk of nephrolithiasis}
    post-flight, with distal convoluted tubule (DCT) ion-transport dysregulation
    strongly implicated~[1]. Standard differential expression (DE) detects only
    mean expression shifts---\textbf{regulatory rewiring} (changes in
    co-expression architecture) is not captured when mean levels are stable.

    \smallskip
    \textbf{Central question:} Does spaceflight restructure the kidney
    co-expression network in an age- and arm-dependent manner, producing
    transcriptomic adaptations that escape DE?

    \smallskip
    We developed a multi-phase pipeline integrating cell-type deconvolution
    (MuSiC), sample-specific network inference (LIONESS), graph embedding
    (node2vec with Procrustes alignment), and permutation testing applied to
    \textbf{RRRM-2 (OSD-771)}---the largest full-factorial spaceflight kidney
    RNA-seq dataset ($n = 80$).
  \end{block}

  \begin{block}{Experimental Design: OSD-771 (RRRM-2)}
    \vspace{0.5ex}
    \renewcommand{\arraystretch}{1.3}
    \begin{center}
    \begin{tabular}{r!{\color{myblue}\vrule width 2pt}l}
      \textbf{Factor} & \textbf{Levels} \\[2pt]
      \hline
      Age   & Young (16 wk) \textbf{vs} Old (34 wk) \\
      Arm   & ISS-T (terminal) \textbf{vs} LAR (live return) \\
      Group & FLT \quad GC \quad VIV \quad BSL \\
      \hline
      Per-cell $n$ & 5 mice $\times$ 2 ages $= 10$ per cell \\
      Total $n$    & \textbf{80} female C57BL/6NTac bulk kidney RNA-seq \\
    \end{tabular}
    \end{center}

    \smallskip
    \textbf{Four primary contrasts (FLT$-$GC):} ISS-T Young,\; ISS-T Old,\;
    LAR Young,\; LAR Old. Arm stratification enables classification of rewiring
    as \emph{ephemeral} (ISS-T-only) vs.\ \emph{persistent} (both arms).
  \end{block}

  %% ---- CONCLUSIONS ----
  \begin{alertblock}{Conclusions}
    \vspace{0.5ex}
    \begin{enumerate}\setlength\itemsep{0.4ex}
      \item Spaceflight induces age-stratified co-expression rewiring
        orthogonal to DE ($\rho \approx 0.03$, $R^2 < 0.3\%$)
      \item \textbf{55--68 silent shifters per contrast}---connectivity
        restructuring with no detectable expression change
      \item Young ISS-T: ECM/complement rewiring;\;
        Old ISS-T: ribosomal/proteostasis disruption
      \item Retinoic acid and PPAR signaling are enriched in LAR recovery
      \item Network topology (AUC $= 0.525$) and expression
        (AUC $= 0.598$) classifiers confirm orthogonality of the two
        representations
      \item Co-expression network analysis is a complementary modality for
        space biology, resolving biology not accessible through DE alone
    \end{enumerate}
  \end{alertblock}

\begin{column}{\colwidth}

  \begin{block}{Genome-Wide Rewiring Landscape}
    \vspace{0.5ex}
    Spaceflight remodels co-expression architecture across all four contrasts.
    Mean rewiring $\Delta \approx 0.27$--$0.29$ (cosine distance); maximum
    single-gene rewiring exceeds $0.53$.

    \smallskip
    \begin{center}
    \small
    \begin{tabular}{lcc}
      \toprule
      \textbf{Contrast (FLT$-$GC)} & \textbf{Mean $\Delta$} & \textbf{Max $\Delta$} \\
      \midrule
      ISS-T Young & 0.284 & 0.515 \\
      ISS-T Old   & 0.287 & 0.511 \\
      LAR Young   & 0.271 & 0.513 \\
      LAR Old     & 0.275 & 0.539 \\
      \bottomrule
    \end{tabular}
    \end{center}

    \smallskip
    \textbf{Orthogonality:} Rewiring $\Delta$ is independent of
    $|\log_2\text{FC}|$ across all four contrasts ($\rho = 0.02$--$0.05$,
    $R^2 < 0.3\%$), confirming that network rewiring captures a dimension of
    spaceflight biology orthogonal to differential expression.

    \vspace{0.8ex}
    \begin{figure}
      \centering
      % FILE: rewiring_vs_DE.png
      \includegraphics[width=0.95\linewidth]{figures/publication/rewiring_vs_logfc.png}
      \caption{\small Rewiring magnitude ($\Delta$) vs.\ $|\log_2\text{FC}|$
        for 2,500 genes across four contrasts. Near-zero Spearman $\rho$
        ($R^2 < 0.3\%$) confirms statistical independence of network rewiring
        from mean expression change.}
    \end{figure}
  \end{block}

  \begin{alertblock}{Silent Shifters: Invisible Transcriptomic Restructuring}
    \vspace{0.5ex}
    \textbf{Silent shifters}---genes in the top-decile of rewiring with
    $|\log_2\text{FC}| < 0.3$ and DE FDR $> 0.2$---are invisible to standard
    DE analysis. The pipeline identifies \textbf{55--68 silent shifters per
    contrast}:

    \smallskip
    \begin{center}\small
    \begin{tabular}{lc}
      \toprule
      \textbf{Contrast} & \textbf{Silent Shifters} \\
      \midrule
      ISS-T Young & 66 \\
      ISS-T Old   & 63 \\
      LAR Young   & 55 \\
      LAR Old     & 68 \\
      \bottomrule
    \end{tabular}
    \end{center}

    \vspace{0.8ex}
    \begin{figure}
      \centering
      % FILE: quadrant_plot.png
      \includegraphics[width=0.95\linewidth]{figures/phase5_silent_shifters/quadrant_plot.png}
      \caption{\small Rewiring score vs.\ $-\log_{10}(q)$ for top-decile genes
        (four contrasts). Dashed lines: 75th-percentile rewiring threshold (red)
        and $q = 0.05$ (green). Genes in the upper-left quadrant have high
        rewiring and low statistical significance---the silent shifter class.}
    \end{figure}
  \end{alertblock}

  \begin{block}{Topological Hub Hits via Focused Permutation}
    \vspace{0.5ex}
    Focused permutation ($K = 50{,}000$; BH correction restricted to top
    decile, $n = 250$) identified \textbf{11 significantly rewired hub genes}
    in ISS-T Young (FDR $< 0.05$):

    \smallskip
    \begin{center}
    \small
    \begin{tabular}{lrl}
      \toprule
      \textbf{Gene} & \textbf{$q$} & \textbf{Biological Role} \\
      \midrule
      Slc6a20a  & 0.025 & Renal amino acid transporter (proximal tubule) \\
      Trf       & 0.025 & Iron trafficking / secretory protein \\
      Fabp7     & 0.033 & Lipid metabolism \\
      Azgp1     & 0.037 & Lipid catabolism \\
      Foxq1     & 0.043 & Epithelial differentiation \\
      Slc22a28  & 0.043 & Organic anion transporter (proximal tubule) \\
      Adgrg4    & 0.044 & Cell--cell communication \\
      Gc        & 0.044 & Vitamin D / mineral homeostasis \\
      Cry2      & 0.050 & Core circadian clock repressor \\
      Pfkp      & 0.050 & Carbohydrate metabolism \\
      \bottomrule
    \end{tabular}
    \end{center}

    \smallskip
    \textbf{ISS-T Old:} Prlr ($q = 0.025$) and Nr4a1 ($q = 0.025$) implicate
    hormonal and stress-response pathways. Edge-wise regression (all 80 samples)
    identified 154 genes at flight-effect FDR $< 0.05$; 25 genes with
    significant age$\times$flight interaction.
  \end{block}

\end{column}

\begin{column}{\colwidth}

  \begin{block}{Age-Dependent Rewiring Divergence}
    \vspace{0.5ex}
    ISS-T rewiring is correlated across age groups (Spearman $\rho = 0.855$),
    but a subset of genes shows age-divergent topological responses. The top
    age-divergent genes include Avpi1, Cxcl14, Parm1, Ak5, and Mcm3, spanning
    ECM remodeling, immune signaling, and cell-cycle regulation.

    \vspace{0.8ex}
    \begin{figure}
      \centering
      % FILE: age_dependent_rewiring.png
      \includegraphics[width=0.95\linewidth]{figures/publication/age_comparison.png}
      \caption{\small \textit{Left:} ISS-T Old vs.\ Young rewiring scatter
        ($\rho = 0.855$). Color encodes age-divergent rewiring magnitude;
        labeled outliers are the most age-divergent genes. \textit{Right:}
        Top 20 age-divergent genes, directional rewiring bar chart.}
    \end{figure}
  \end{block}

  \begin{block}{Pathway Remodeling: Age-Dependent Signatures}
    \vspace{0.5ex}
    Gene set enrichment (Fisher's exact, BH-corrected; KEGG / Reactome /
    Hallmark / WikiPathways) reveals contrast-specific pathway signatures:

    \smallskip
    \noindent\textbf{ISS-T Old} (27 pathways, FDR $< 0.05$): Ribosome, Peptide
    Chain Elongation, Chaperone-Mediated Autophagy (OR $= 23.6$, $q = 0.005$),
    Selective Autophagy\\
    \textit{$\to$ proteostasis disruption in the aged spaceflight kidney}

    \smallskip
    \noindent\textbf{LAR Old} (22 pathways): PPAR Signaling (OR $= 7.5$,
    $q = 0.027$), NOTCH Signaling, Complement/Coagulation\\
    \textit{$\to$ coordinated metabolic and proteostatic recovery upon return}

    \smallskip
    \noindent\textbf{LAR Young} (11 pathways): PPAR Signaling (OR $= 10.1$,
    $q = 0.002$): Acaa1b, Acox2, Adipoq, Angptl4, Fabp5, Hmgcs2, Scd1\\
    \textit{$\to$ PPAR-centered metabolic recovery program}

    \vspace{0.8ex}
    \begin{figure}
      \centering
      % FILE: pathway_dotplot.png
      \includegraphics[width=0.95\linewidth]{figures/publication/pathway_dotplot.png}
      \caption{\small Pathway enrichment bubble dot plot across four contrasts
        (top-decile Fisher's exact, BH-corrected). Bubble size $=$ hit gene
        count; color intensity $= -\log_{10}(q_\text{BH})$.}
    \end{figure}
  \end{block}

  \begin{block}{Recovery \& Persistence of Rewiring}
    \vspace{0.5ex}
    Approximately 15\% of top-decile rewiring genes are ISS-T-specific
    (ephemeral), resolving after return to Earth. Another 22\% show persistent
    rewiring in both arms.

    \vspace{0.8ex}
    \begin{figure}
      \centering
      % FILE: arm_comparison.png
      \includegraphics[width=0.95\linewidth]{figures/publication/arm_comparison.png}
      \caption{\small ISS-T vs.\ LAR rewiring scatter for young (left) and old
        (right) animals ($\rho \approx 0.86$). Color encodes ISS-T $-$ LAR
        rewiring difference. Genes above the diagonal are ISS-T-specific;
        genes below are LAR-amplified.}
    \end{figure}
  \end{block}

  \begin{exampleblock}{Convergent Signal: Retinoic Acid Metabolism}
    \vspace{0.5ex}
    {\small
    Retinoic acid emerges as \textbf{a reproducible finding} across orthogonal
    inference strategies:
    \begin{itemize}\setlength\itemsep{0.3ex}
      \item \textbf{Pathway:} Retinol Metabolism---strongest enrichment in LAR
        Young (OR $= 19.9$, $q = 1.6\times10^{-5}$); 9/35 constituent genes
        in the top rewiring decile
      \item \textbf{Gene:} Ugt1a1 ($q = 0.056$), Ugt1a10 ($q = 0.080$),
        Cyp4a12b ($q = 0.080$) approach significance by focused permutation
      \item \textbf{Arm specificity:} Signal peaks in LAR (post-return),
        consistent with retinoic acid as a recovery response rather than a
        direct microgravity effect
    \end{itemize}

    \vspace{0.5ex}
    \begin{figure}
      \centering
      % FILE: retinol_subnetwork.png
      \includegraphics[width=0.95\linewidth]{figures/publication/retinol_subnetwork.png}
      \caption{\small \textit{Left:} Per-gene retinol metabolism rewiring
        heatmap across all four contrasts. \textit{Right:} Mean rewiring of
        retinol pathway genes vs.\ genome background by contrast.}
    \end{figure}
    }
  \end{exampleblock}
